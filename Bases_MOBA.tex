\section{Formato del Torneo}
\begin{itemize}
    \item \textbf{Modo de Juego:} Personalizada - Modo Torneo (Draft Pick 5v5).
    \item \textbf{Estructura de Brackets:} Las rondas eliminatorias y cuartos de final se jugarán al Mejor de 1 (BO1). Las semifinales y la Gran Final se jugarán al Mejor de 3 (BO3).
    \item \textbf{Dispositivos:} Única y estrictamente dispositivos móviles (celulares o tablets). El uso de emuladores (PC) está totalmente prohibido.
    \item \textbf{Condición de Victoria:} La partida finaliza cuando el Nexo (Base) enemigo es destruido o mediante la rendición oficial de un equipo a través del sistema del juego.
\end{itemize}

\section{Inscripciones y Requisitos}
\begin{itemize}
    \item \textbf{Estructura del Equipo:} Cada escuadra debe registrar obligatoriamente a 5 jugadores titulares. Opcionalmente, se permite el registro de hasta 2 jugadores suplentes.
    \item \textbf{Identificación (IDs):} Es obligatorio registrar tanto el \textbf{ID de Cuenta} como el \textbf{ID de Servidor} (los 4 dígitos entre paréntesis) de todos los jugadores.
    \item \textbf{El Capitán:} Será el único medio de comunicación oficial. Debe proveer su nombre real, número de WhatsApp y su institución (UNAP u otra).
    \item \textbf{Validación de Pagos (Yape):} La inscripción será válida tras verificar el comprobante de pago. El pozo de premios (Prize Pool) será transferido única y exclusivamente a la cuenta del capitán.
\end{itemize}

\section{Gestión de Salas (Lobby) y Seguridad}
\begin{itemize}
    \item \textbf{Creación de la Sala:} El árbitro creará la sala e invitará a los dos capitanes. Los capitanes son responsables de invitar al resto de sus escuadras a sus respectivos lados (Equipo Rojo o Azul).
    \item \textbf{Tolerancia y Puntualidad:} Se otorgarán exactamente 10 minutos de tolerancia para que ambos equipos completen su alineación de 5 jugadores en la sala. Pasado el tiempo, el equipo incompleto será descalificado (W.O.).
    \item \textbf{Coincidencia de Nicknames:} Solo podrán participar los jugadores cuyo Nickname e ID coincidan exactamente con la base de datos oficial. Jugadores no registrados serán expulsados de la sala.
\end{itemize}

\section{Reglas de Partida (Draft Pick y Pausas)}
\begin{itemize}
    \item \textbf{Desconexiones en el Draft Pick:} Si un jugador se desconecta del lobby \textit{antes} de que ambos equipos bloqueen (baneen) a su primer héroe, la sala será reiniciada. Si la desconexión ocurre \textit{después} del primer baneo, la fase de selección debe continuar y la partida se jugará obligatoriamente.
    \item \textbf{Sistema de Pausas:} Cada escuadra tiene derecho a solicitar un máximo de \textbf{2 pausas} por partida por motivos de conexión (lag).
    \item \textbf{Duración de Pausas:} Ninguna pausa podrá exceder los 3 minutos.
    \item \textbf{Procedimiento:} El capitán o jugador afectado debe solicitar la pausa escribiendo en el chat global ("All"). El árbitro será el único encargado de ejecutar la pausa en el sistema.
\end{itemize}

\section{Conducta y Sanciones (Tolerancia Cero)}
\begin{itemize}
    \item \textbf{Uso del Suplente:} Solo pueden ingresar antes de que inicie la fase de Draft Pick. Prohibido realizar cambios de jugadores durante una partida en curso.
    \item \textbf{Filtración de Contraseñas:} Compartir capturas, enlaces o el ID de la sala privada con personas ajenas al torneo resultará en la \textbf{descalificación automática} del equipo, sin derecho a reembolso.
    \item \textbf{Comportamiento Tóxico:} El uso de insultos, "recall spam" malintencionado extremo o agresión verbal en el chat global hacia los árbitros o rivales resultará en un "Warning" (Advertencia) o descalificación directa, a discreción de los organizadores.
\end{itemize}
